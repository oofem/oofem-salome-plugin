## In Salome 9.15 (and 9.16), the SMESH CORBA server is not fully activated until you switch modules.

import salome

entry = "0:1:2:6"

mesh = salome.IDToObject(entry)

print("Mesh:", mesh.GetName())
print("Nodes:", mesh.NbNodes())
print("Elements:", mesh.NbElements())

groups = mesh.GetGroups()
for g in groups:
    print("Group:", g.GetName(), "Entry:", salome.ObjectToID(g))

    node_ids = g.GetIDs()
    print("Node IDs:", node_ids)

    elem_ids = g.GetElementIDs()
    print("Element IDs:", elem_ids)





import salome.smesh.smeshstudytools as st
tools = st.SMeshStudyTools()
mesh_obj = tools.getMeshObjectSelected()
if (mesh_obj):
    print("Selected mesh name:", mesh_obj.GetName())
